(function() { const grid = document.getElementById('grid'); const wrapper = document.getElementById('wrapper'); const COLS = 18; const ROWS = 14; // ===== ЗАГРУЗКА ТАЙЛОВ ===== for (let row = 0; row < ROWS; row++) { for (let col = 1; col <= COLS; col++) { const img = document.createElement('img'); img.src = `https://raw.githubusercontent.com/chuk0vskii/CHERTOGI_MAP/main/tiles4/tile_${row}_${col}.jpg`; img.alt = `tile_${row}_${col}`; grid.appendChild(img); } } // ===== РАЗМЕР ТАЙЛОВ ===== let tileSize = 0; function fitTiles() { const rect = wrapper.getBoundingClientRect(); const tileW = Math.floor(rect.width / COLS); const tileH = Math.floor(rect.height / ROWS); tileSize = Math.min(tileW, tileH); const images = grid.querySelectorAll('img'); images.forEach(img => { img.style.width = tileSize + 'px'; img.style.height = tileSize + 'px'; }); const gridW = tileSize * COLS; const gridH = tileSize * ROWS; grid.style.width = gridW + 'px'; grid.style.height = gridH + 'px'; centerMap(); } // ===== СОСТОЯНИЕ ===== let scale = 1; let tx = 0; let ty = 0; let minScale = 1; const MAX_SCALE = 2.0; const STEP = 0.08; function updateTransform() { const transform = `translate(${tx}px, ${ty}px) scale(${scale})`; grid.style.transform = transform; } function centerMap() { const rect = wrapper.getBoundingClientRect(); const gridW = tileSize * COLS; const gridH = tileSize * ROWS; const sx = rect.width / gridW; const sy = rect.height / gridH; minScale = Math.min(sx, sy); scale = minScale; tx = (rect.width - gridW * scale) / 2; ty = (rect.height - gridH * scale) / 2; updateTransform(); } // ===== ЗУМ ПОД КУРСОРОМ ===== wrapper.addEventListener('wheel', function(e) { e.preventDefault(); const rect = wrapper.getBoundingClientRect(); const mouseX = e.clientX - rect.left; const mouseY = e.clientY - rect.top; const delta = e.deltaY > 0 ? -STEP : STEP; let newScale = scale + delta; newScale = Math.min(Math.max(newScale, minScale), MAX_SCALE); if (newScale === scale) return; // Точка под курсором в координатах карты (до зума) const px = (mouseX - tx) / scale; const py = (mouseY - ty) / scale; scale = newScale; // Новое смещение, чтобы точка под курсором осталась на месте tx = mouseX - px * scale; ty = mouseY - py * scale; // Ограничения (чтобы карта не уезжала за край) const maxX = rect.width - tileSize * COLS * scale; const maxY = rect.height - tileSize * ROWS * scale; tx = Math.min(Math.max(tx, maxX), 0); ty = Math.min(Math.max(ty, maxY), 0); updateTransform(); }, { passive: false }); // ===== ПЕРЕТАСКИВАНИЕ ===== let isDragging = false; let startX, startY, startTx, startTy; wrapper.addEventListener('mousedown', function(e) { if (e.button !== 0) return; isDragging = true; startX = e.clientX; startY = e.clientY; startTx = tx; startTy = ty; wrapper.style.cursor = 'grabbing'; }); document.addEventListener('mousemove', function(e) { if (!isDragging) return; tx = startTx + (e.clientX - startX); ty = startTy + (e.clientY - startY); updateTransform(); }); document.addEventListener('mouseup', function() { if (isDragging) { isDragging = false; wrapper.style.cursor = 'grab'; } }); // ===== ЗАПУСК ===== fitTiles(); window.addEventListener('resize', function() { fitTiles(); centerMap(); }); setTimeout(centerMap, 500); })();
