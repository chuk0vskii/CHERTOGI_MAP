// ===== ЗУМ ПОД КУРСОРОМ (С ПРАВИЛЬНЫМ МИНИМУМОМ) ===== wrapper.addEventListener('wheel', function(e) { e.preventDefault(); // Координаты мыши внутри wrapper const rect = wrapper.getBoundingClientRect(); const mouseX = e.clientX - rect.left; const mouseY = e.clientY - rect.top; // Новый масштаб const delta = e.deltaY > 0 ? -0.08 : 0.08; let newScale = scale + delta; // ===== ГЛАВНОЕ ИЗМЕНЕНИЕ: НЕ ДАЁМ СТАТЬ МЕНЬШЕ 1.0 ===== newScale = Math.min(Math.max(newScale, 1.0), 2.0); if (newScale === scale) return; // Процентное положение мыши внутри картинки const percentX = (mouseX - translateX) / (IMG_WIDTH * scale); const percentY = (mouseY - translateY) / (IMG_HEIGHT * scale); // Новые размеры const newWidth = IMG_WIDTH * newScale; const newHeight = IMG_HEIGHT * newScale; // Новое смещение, чтобы точка под мышью осталась на месте translateX = mouseX - percentX * newWidth; translateY = mouseY - percentY * newHeight; // Ограничения const wrapperRect = wrapper.getBoundingClientRect(); const maxX = wrapperRect.width - newWidth; const maxY = wrapperRect.height - newHeight; translateX = Math.min(Math.max(translateX, maxX), 0); translateY = Math.min(Math.max(translateY, maxY), 0); scale = newScale; updateTransform(); }, { passive: false });
